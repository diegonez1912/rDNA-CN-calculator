## BASH SCRIPT FOR ONLY rDNA CYTOBANDS IN EACH CHR ## 

python -c "
import pandas as pd
import sys

# Load data
df = pd.read_csv('OV_ploidy.txt', sep='\t')

# 1. Define SPECIFIC Target Bands (No buffers)
# 45S is strictly in the p12 band of acrocentrics
targets45s = r'p12' 
# 5S is strictly in the q42 band of Chr 1
target5s = r'q42'

# 2. Create Masks
acro_chrs = ('13', '14', '15', '21', '22')

mask45 = df['Cytoband'].str.startswith(acro_chrs, na=False) & \
         df['Cytoband'].str.contains(targets45s, na=False)

mask5 = df['Cytoband'].str.startswith('1', na=False) & \
        df['Cytoband'].str.contains(target5s, na=False)

# Check counts
n45 = mask45.sum()
n5 = mask5.sum()
print(f'Specific genes found for 45S (p12): {n45}')
print(f'Specific genes found for 5S (q42): {n5}')

if n45 == 0 or n5 == 0:
    print('Warning: One of the regions has 0 genes. The All average might be skewed.')

# 3. Calculate Means
# Skip first 3 metadata columns
c45 = df[mask45].iloc[:, 3:].mean()
c5 = df[mask5].iloc[:, 3:].mean()

# Calculate All as the balanced average of the two regions
# This prevents 5S from dominating the 45S signal
c_all = (c45 + c5) / 2

# 4. Combine and Export
master = pd.DataFrame({
    'SampleID': c45.index,
    'C_rDNA_All': c_all.values,
    'C_45S': c45.values,
    'C_5S': c5.values
})

# Save with a new name to distinguish from the buffered version
output_name = 'OV_ploidy_specific_cytobands.tsv'
master.to_csv(output_name, sep='\t', index=False)
print(f'Success: File written to {output_name}')
"


## BASH SCRIPT FOR rDNA AND THEIR ADJACENT CYTOBANDS IN EACH CHR ## 

python -c "
import pandas as pd
import sys

# Load data
df = pd.read_csv('GBM_ploidy.txt', sep='\t')

# 1. Define Target Bands
# We use regex patterns to catch both the main band and any sub-bands (e.g., 1q42.1)
# 45S: Focus on the p-arm (short arm) of acrocentrics
targets45s = r'p11|p12|p13|pter' 
# 5S: Focus on the 1q41-43 region
target5s = r'1q41|1q42|1q43'

# 2. Create Masks
acro_chrs = ('13', '14', '15', '21', '22')

# Filter for Acrocentric p-arms
mask45 = df['Cytoband'].str.startswith(acro_chrs, na=False) & \
         df['Cytoband'].str.contains(targets45s, na=False)

# Filter for Chr 1 cluster
mask5 = df['Cytoband'].str.startswith('1', na=False) & \
        df['Cytoband'].str.contains(target5s, na=False)

mask_all = mask45 | mask5

# Check if we actually found any genes
n45 = mask45.sum()
n5 = mask5.sum()
print(f'Genes found for 45S (Chr 13,14,15,21,22 p-arms): {n45}')
print(f'Genes found for 5S (Chr 1q41-43): {n5}')

if (n45 + n5) == 0:
    print('Error: No genes found matching these cytobands. Check file content.')
    sys.exit(1)

# 3. Calculate Means
# Starting from index 3 (skipping Gene Symbol, Locus ID, Cytoband)
c_all = df[mask_all].iloc[:, 3:].mean()
c45 = df[mask45].iloc[:, 3:].mean()
c5 = df[mask5].iloc[:, 3:].mean()

# 4. Combine and Export
master = pd.DataFrame({
    'SampleID': c_all.index,
    'C_rDNA_All': c_all.values,
    'C_45S': c45.values,
    'C_5S': c5.values
})

output_name = 'GBM_ploidy_adjacent_cytoband.tsv'
master.to_csv(output_name, sep='\t', index=False)
print(f'Success: File written to {output_name}')
"

import pandas as pd
import sys

df = pd.read_csv('GBM_ploidy.txt', sep='\t')

# 1. Best available proxies for rDNA locations
# 45S: Use p11 or q11, q12, q13 (the regions flanking the rDNA/Centromere)
targets45s = r'p11|q11|q12|q13' 
# 5S: Strictly the q42 region
target5s = r'q42'

acro_chrs = ('13', '14', '15', '21', '22')

# 2. Create Masks
mask45 = df['Cytoband'].str.startswith(acro_chrs, na=False) & \
         df['Cytoband'].str.contains(targets45s, na=False)
mask5 = df['Cytoband'].str.startswith('1', na=False) & \
        df['Cytoband'].str.contains(target5s, na=False)

n45 = mask45.sum()
n5 = mask5.sum()
print(f'Genes found for 45S Proxy (p11/q11/q12/q13): {n45}')
print(f'Genes found for 5S (q42): {n5}')

# 3. Calculate Means
c45 = df[mask45].iloc[:, 3:].mean()
c5 = df[mask5].iloc[:, 3:].mean()

# Balanced average so 5S (150 genes) doesn't drown out 45S
c_all = (c45 + c5) / 2

# 4. Export
master = pd.DataFrame({
    'SampleID': c45.index,
    'C_rDNA_All': c_all.values,
    'C_45S': c45.values,
    'C_5S': c5.values
})

output_name = 'GBM_ploidy_proxy.tsv'
master.to_csv(output_name, sep='\t', index=False)
print(f'Success: File written to {output_name}')
"


## BASH SCRIPT FOR CREATING csv FILE FOR RSTUDIO ## 

python -c "
import pandas as pd

# Load both datasets
proxy = pd.read_csv('GBM_ploidy_proxy.tsv', sep='\t')
adj = pd.read_csv('GBM_ploidy_adjacent_cytoband.tsv', sep='\t')

# Rename columns to distinguish them
proxy = proxy.rename(columns={'C_45S': '45S_Proxy', 'C_5S': '5S_Proxy', 'C_rDNA_All': 'All_Proxy'})
adj = adj.rename(columns={'C_45S': '45S_Adj', 'C_5S': '5S_Adj', 'C_rDNA_All': 'All_Adj'})

# Merge on SampleID
master = pd.merge(proxy, adj[['SampleID', '45S_Adj', '5S_Adj']], on='SampleID')

# Clean ShortID for R
master['ShortID'] = master['SampleID'].str[:12]

output_name = 'GBM_rDNA_Comparison.csv'
master.to_csv(output_name, index=False)
print(f'Master comparison file created: {output_name}')
"
