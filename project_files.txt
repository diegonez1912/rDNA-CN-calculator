

## BEFORE RUNNING calc_rDNA_depth.sh (BOTH OF THEM) ##
    - Create log folder in "logs"


For multiple runs with GERM-LINE SAMPLES, run as:

for bam in /home/diegoonez/CancerEvolution/Datasets/TCGA_WGS/randiip/data/TCGA_GermLine/TCGA-TGCT/*.bam; do
    sample=$(basename "$bam" .bam)
        
    echo "Submitting: ${sample}"
        
    sbatch --job-name="TGCT_GL_${sample}" \
        --open-mode=append \
        -o logs/logs_TGCT_GL/${sample}.log \
        -e logs/logs_TGCT_GL/${sample}.log \
        scripts/calc_rDNA_depth_GL.sh "$bam" "TGCT_GL"
    
done

for log in $(grep -L "DONE" logs/logs_BRCA_TUMOR/*.log); do

    echo "Cleaning up intermediate files for: $id"
    rm -f output/BRCA_TUMOR/"$id".rDNA_onRef.sorted.bam \
          output/BRCA_TUMOR/"$id".rDNA_onRef.sorted.bam.bai \

done
